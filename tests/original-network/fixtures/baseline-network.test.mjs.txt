const SWF_HASH = 'a'.repeat(64);
const CREATE = { type: 'create', swfHash: SWF_HASH, mode: 'Coop', map: 'Boxy', characters: [0, 3] };
const SETTINGS = { difficulty: 0, speed: 1, devils: 1, collisions: 1, damage: 1, gameAmount: 1 };
const matches = (expected, actual) => Array.isArray(expected) ? expected.includes(actual) : expected === actual;

class Inbox {
  constructor(ws) {
    this.ws = ws;
    this.messages = [];
    this.waiters = [];
    this.closed = new Promise((resolve) => ws.once('close', (code) => resolve(code)));
    ws.on('error', () => {});
    ws.on('message', (data) => {
      const message = JSON.parse(data.toString());
      const index = this.waiters.findIndex((waiter) => matches(waiter.type, message.type));
      if (index < 0) this.messages.push(message);
      else {
        const [waiter] = this.waiters.splice(index, 1);
        clearTimeout(waiter.timer);
        waiter.resolve(message);
      }
    });
  }

  send(message) {
    this.ws.send(JSON.stringify(message));
  }

  next(type, timeout = 2000) {
    const index = this.messages.findIndex((message) => matches(type, message.type));
    if (index >= 0) return Promise.resolve(this.messages.splice(index, 1)[0]);
    return new Promise((resolve, reject) => {
      const waiter = { type, resolve, timer: undefined };
      waiter.timer = setTimeout(() => {
        const position = this.waiters.indexOf(waiter);
        if (position >= 0) this.waiters.splice(position, 1);
        reject(Object.assign(new Error(`Timed out waiting for ${type}`), { code: 'WAIT_TIMEOUT' }));
      }, timeout);
      this.waiters.push(waiter);
    });
  }

  async absent(type, timeout = 70) {
    await assert.rejects(this.next(type, timeout), { code: 'WAIT_TIMEOUT' });
  }

  async close() {
    if (this.ws.readyState === WebSocket.CLOSED) return;
    this.ws.close();
    await this.closed;
  }
}

async function fixture(t, options = {}) {
  const directory = await mkdtemp(path.join(os.tmpdir(), 'original-network-'));
  const webRoot = path.join(directory, 'web');
  await mkdir(webRoot);
  await writeFile(path.join(webRoot, 'index.html'), '<!doctype html><title>Original lockstep fixture</title>');
  await writeFile(path.join(webRoot, 'engine.wasm'), Buffer.from([0, 97, 115, 109, 1, 0, 0, 0]));
  await writeFile(path.join(directory, 'secret.txt'), 'outside-root-secret');
  const service = createOriginalServer({
    host: '127.0.0.1', port: 0, webRoot, shutdownGraceMs: 300, ...options,
  });
  t.after(async () => {
    await service.close();
    await rm(directory, { recursive: true, force: true });
  });
  await service.listen();
  const address = service.address();
  const origin = `http://127.0.0.1:${address.port}`;
  return { service, directory, webRoot, address, origin, url: `ws://127.0.0.1:${address.port}/original-ws` };
}

// Pin the prior32/64 transport configuration for every existing regression.
// New default64/128 cases below deliberately use fixture() directly.
async function fixture32(t, options = {}) {
  return fixture(t, { maxReplayInFlightFrames: 64, maxReplayBatchFrames: 32, ...options });
}

async function connect(t, fixture) {
  const ws = new WebSocket(fixture.url, { origin: fixture.origin });
  const inbox = new Inbox(ws);
  t.after(() => inbox.close());
  await once(ws, 'open');
  return inbox;
}

async function pair(t, fixture, create = CREATE) {
  const host = await connect(t, fixture);
  host.send(create);
  const invite = await host.next('joined');
  assert.equal(invite.slot, 1);
  assert.match(invite.room, /^[A-Za-z0-9_-]{12}$/);
  assert.match(invite.token, /^[A-Za-z0-9_-]{43}$/);
  const guest = await connect(t, fixture);
  guest.send({ type: 'join', room: invite.room, token: invite.token, swfHash: create.swfHash });
  const joined = await guest.next('joined');
  assert.equal(joined.slot, 2);
  assert.equal(joined.room, invite.room);
  assert.deepEqual(await host.next('peer'), { type: 'peer', slot: 2, connected: true });
  return { host, guest, invite };
}

async function prepare(players) {
  players.host.send({ type: 'start' });
  const [hostPrepare, guestPrepare] = await Promise.all([
    players.host.next('prepare'), players.guest.next('prepare'),
  ]);
  assert.deepEqual(hostPrepare, guestPrepare);
  assert.equal(hostPrepare.swfHash, SWF_HASH);
  assert.equal(hostPrepare.mode, 'Coop');
  assert.equal(hostPrepare.map, 'Boxy');
  assert.deepEqual(hostPrepare.characters, [0, 3]);
  assert.ok(Number.isInteger(hostPrepare.seed) && hostPrepare.seed > 0 && hostPrepare.seed < 2147483647);
  return hostPrepare;
}

async function begin(players) {
  await prepare(players);
  players.host.send({ type: 'input', seq: 1, mask: 17 });
  players.guest.send({ type: 'input', seq: 1, mask: 8 });
  players.host.send({ type: 'ready' });
  players.guest.send({ type: 'ready' });
  const frames = await Promise.all([players.host.next('frame'), players.guest.next('frame')]);
  assert.deepEqual(frames[0], { type: 'frame', tick: 1, p1: 17, p2: 8 });
  assert.deepEqual(frames[0], frames[1]);
  return frames[0];
}

async function beginNeutral(players) {
  await prepare(players);
  players.host.send({ type: 'ready' });
  players.guest.send({ type: 'ready' });
  const first = await players.host.next('frame');
  assert.deepEqual(first, { type: 'frame', tick: 1, p1: 0, p2: 0 });
  assert.deepEqual(await players.guest.next('frame'), first);
}

async function acknowledgeAndNext(players, tick) {
  for (const player of [players.host, players.guest]) player.send({ type: 'ack', tick, digest: `native-${tick}` });
  const frame = await players.host.next('frame');
  assert.equal(frame.tick, tick + 1);
  assert.deepEqual(await players.guest.next('frame'), frame);
  return frame;
}

function request(fixture, target, method = 'GET') {
  return new Promise((resolve, reject) => {
    const outgoing = http.request({
      hostname: '127.0.0.1', port: fixture.address.port, path: target, method,
    }, (response) => {
      const chunks = [];
      response.on('data', (chunk) => chunks.push(chunk));
      response.on('end', () => resolve({
        status: response.statusCode, headers: response.headers, body: Buffer.concat(chunks).toString('utf8'),
      }));
      response.on('error', reject);
    });
    outgoing.on('error', reject);
    outgoing.end();
  });
}

async function deniedJoin(t, fixture, message, code) {
  const client = await connect(t, fixture);
  client.send(message);
  assert.equal((await client.next('error')).code, code);
  assert.equal(await client.closed, 1008);
}

test('invite secrets bind to one room; hashes, capacity and one owned slot are enforced', async (t) => {
  const fixtureValue = await fixture32(t);
  const players = await pair(t, fixtureValue);
  const anotherHost = await connect(t, fixtureValue);
  anotherHost.send(CREATE);
  const otherInvite = await anotherHost.next('joined');
  assert.notEqual(otherInvite.token, players.invite.token);
  await deniedJoin(t, fixtureValue, {
    type: 'join', room: players.invite.room, token: otherInvite.token, swfHash: SWF_HASH,
  }, 'join_denied');
  await deniedJoin(t, fixtureValue, {
    type: 'join', room: players.invite.room, token: players.invite.token, swfHash: 'b'.repeat(64),
  }, 'swf_mismatch');
  await deniedJoin(t, fixtureValue, {
    type: 'join', room: players.invite.room, token: players.invite.token, swfHash: SWF_HASH,
  }, 'room_full');
  assert.equal(fixtureValue.service.stats().rooms, 2);
  await begin(players);
});

test('live room directory exposes only opted-in free compatible rooms without invite secrets', async (t) => {
  const fixtureValue = await fixture32(t);
  const privateHost = await connect(t, fixtureValue);
  privateHost.send(CREATE);
  const privateInvite = await privateHost.next('joined');
  const incompatibleHost = await connect(t, fixtureValue);
  incompatibleHost.send({ ...CREATE, swfHash: 'b'.repeat(64), listed: true, name: 'Other version' });
  const incompatibleInvite = await incompatibleHost.next('joined');

  const watcher = await connect(t, fixtureValue);
  watcher.send({ type: 'list', swfHash: SWF_HASH });
  assert.deepEqual(await watcher.next('rooms'), { type: 'rooms', rooms: [] });

  const host = await connect(t, fixtureValue);
  host.send({ ...CREATE, listed: true, name: 'Комната друзей' });
  const invite = await host.next('joined');
  const listed = await watcher.next('rooms');
  assert.deepEqual(listed, { type: 'rooms', rooms: [{
    room: invite.room, name: 'Комната друзей', mode: 'Coop', map: 'Boxy', players: 1, capacity: 2,
  }] });
  const publicText = JSON.stringify(listed);
  for (const secret of [privateInvite.token, incompatibleInvite.token, invite.token]) {
    assert.equal(publicText.includes(secret), false, 'room directory must never contain an invitation secret');
  }
  assert.equal(publicText.includes(privateInvite.room), false);
  assert.equal(publicText.includes(incompatibleInvite.room), false);

  watcher.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH });
  assert.equal((await watcher.next('joined')).slot, 2);
  assert.deepEqual(await host.next('peer'), { type: 'peer', slot: 2, connected: true });
  const observer = await connect(t, fixtureValue);
  observer.send({ type: 'list', swfHash: SWF_HASH });
  assert.deepEqual(await observer.next('rooms'), { type: 'rooms', rooms: [] });
  host.send({ type: 'start' });
  assert.deepEqual(await host.next('prepare'), await watcher.next('prepare'));
  assert.deepEqual(await observer.next('rooms'), { type: 'rooms', rooms: [] });
  await watcher.absent('rooms', 30);
});

test('simultaneous open-room clicks claim one slot and keep the losing browser usable', async (t) => {
  const fixtureValue = await fixture32(t);
  const host = await connect(t, fixtureValue);
  host.send({ ...CREATE, listed: true });
  const invite = await host.next('joined');
  const first = await connect(t, fixtureValue);
  const second = await connect(t, fixtureValue);
  for (const client of [first, second]) client.send({ type: 'list', swfHash: SWF_HASH });
  for (const client of [first, second]) {
    const { rooms } = await client.next('rooms');
    assert.equal(rooms.length, 1);
    assert.equal(rooms[0].name, 'Boxhead Coop');
  }
  for (const client of [first, second]) client.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH });
  const outcomes = await Promise.all([first.next(['joined', 'join_failed']), second.next(['joined', 'join_failed'])]);
  assert.equal(outcomes.filter((message) => message.type === 'joined').length, 1);
  assert.equal(outcomes.filter((message) => message.type === 'join_failed').length, 1);
  const loserIndex = outcomes.findIndex((message) => message.type === 'join_failed');
  const loser = [first, second][loserIndex];
  assert.deepEqual(outcomes[loserIndex], { type: 'join_failed', room: invite.room, code: 'unavailable' });
  assert.deepEqual(await host.next('peer'), { type: 'peer', slot: 2, connected: true });
  assert.deepEqual(await loser.next('rooms'), { type: 'rooms', rooms: [] });
  assert.equal(loser.ws.readyState, WebSocket.OPEN);
  await host.absent('peer', 25);

  // The list can contain a second empty update from the first successful claim.
  while (loser.messages.some((message) => message.type === 'rooms')) await loser.next('rooms');
  const anotherHost = await connect(t, fixtureValue);
  anotherHost.send({ ...CREATE, listed: true, name: 'Next available game' });
  const anotherInvite = await anotherHost.next('joined');
  assert.equal((await loser.next('rooms')).rooms[0].room, anotherInvite.room);
  loser.send({ type: 'join_open', room: anotherInvite.room, swfHash: SWF_HASH });
  assert.equal((await loser.next('joined')).room, anotherInvite.room);
  assert.deepEqual(await anotherHost.next('peer'), { type: 'peer', slot: 2, connected: true });
  assert.equal(fixtureValue.service.stats().rooms, 2);
});

test('a configured one-frame window waits for both identical ACKs and preserves pacing', async (t) => {
  const fixtureValue = await fixture32(t, { maxInFlightFrames: 1 });
  const players = await pair(t, fixtureValue);
  await prepare(players);
  players.host.send({ type: 'input', seq: 0, mask: 17 });
  players.guest.send({ type: 'input', seq: 0, mask: 8 });
  players.host.send({ type: 'ready' });
  await players.host.absent('frame');
  await players.guest.absent('frame', 20);
  players.guest.send({ type: 'ready' });
  const first = await players.host.next('frame');
  assert.deepEqual(first, { type: 'frame', tick: 1, p1: 17, p2: 8 });
  assert.deepEqual(await players.guest.next('frame'), first);
  players.host.send({ type: 'ack', tick: 1, digest: 'native-world-1' });
  players.host.send({ type: 'input', seq: 1, mask: 2 });
  await players.host.absent('frame');
  players.guest.send({ type: 'ack', tick: 1, digest: 'native-world-1' });
  const second = await players.host.next('frame');
  const receivedSecond = performance.now();
  assert.deepEqual(second, { type: 'frame', tick: 2, p1: 2, p2: 8 });
  assert.deepEqual(await players.guest.next('frame'), second);
  players.host.send({ type: 'ack', tick: 2, digest: 'native-world-2' });
  players.guest.send({ type: 'ack', tick: 2, digest: 'native-world-2' });
  await players.host.absent('frame', 20);
  const third = await players.host.next('frame');
  assert.equal(third.tick, 3);
  assert.deepEqual(await players.guest.next('frame'), third);
  assert.ok(performance.now() - receivedSecond >= 30, 'loopback receipt interval should reflect the 40 ms server pacing');
});

test('default two-frame live window maintains cadence without ACKs and bounds a slow peer', async (t) => {
  const fixtureValue = await fixture32(t);
  const players = await pair(t, fixtureValue);
  await begin(players);
  const [hostSecond, guestSecond] = await Promise.all([players.host.next('frame'), players.guest.next('frame')]);
  assert.equal(hostSecond.tick, 2);
  assert.deepEqual(hostSecond, guestSecond);
  // One fast participant alone must not release any window capacity.
  for (let tick = 1; tick <= 2; tick += 1) players.guest.send({ type: 'ack', tick, digest: `native-${tick}` });
  await players.host.absent('frame');
  players.host.send({ type: 'input', seq: 2, mask: 2 });
  players.host.send({ type: 'ack', tick: 1, digest: 'native-1' });
  const third = await players.host.next('frame');
  assert.deepEqual(third, { type: 'frame', tick: 3, p1: 2, p2: 8 });
  assert.deepEqual(await players.guest.next('frame'), third);
  await players.host.absent('frame');
});

test('original settings reach both clients and select 12.5, 25 or 50 Hz pacing', async (t) => {
  for (const [speed, frameMs] of [80, 40, 20].entries()) {
    await t.test(`speed ${speed}`, async (subtest) => {
      const fixtureValue = await fixture32(subtest);
      const settings = { ...SETTINGS, speed, difficulty: 3, devils: 0, collisions: 0, damage: 0, gameAmount: 2 };
      const players = await pair(subtest, fixtureValue, { ...CREATE, settings });
      const prepared = await prepare(players);
      assert.deepEqual(prepared.settings, settings);
      assert.equal(prepared.frameMs, frameMs);
      players.host.send({ type: 'ready' });
      players.guest.send({ type: 'ready' });
      await players.host.next('frame');
      const received = performance.now();
      const second = await players.host.next('frame');
      assert.equal(second.tick, 2);
      assert.ok(performance.now() - received >= frameMs - 10, 'game speed must pace emitted frames');
    });
  }
  await t.test('omitted settings use original defaults', async (subtest) => {
    const fixtureValue = await fixture32(subtest);
    const players = await pair(subtest, fixtureValue);
    const prepared = await prepare(players);
    assert.deepEqual(prepared.settings, SETTINGS);
    assert.equal(prepared.frameMs, 40);
  });
});

test('a matched native completion retires outstanding later frames without a false error', async (t) => {
  const fixtureValue = await fixture32(t, { maxInFlightFrames: 8 });
  const players = await pair(t, fixtureValue);
  await begin(players);
  for (let tick = 2; tick <= 8; tick += 1) {
    assert.equal((await players.host.next('frame')).tick, tick);
    assert.equal((await players.guest.next('frame')).tick, tick);
  }
  for (let tick = 1; tick <= 2; tick += 1) {
    for (const player of [players.host, players.guest]) player.send({ type: 'ack', tick, digest: `native-${tick}` });
  }
  players.host.send({ type: 'ack', tick: 3, digest: 'native-final', complete: 'player dead' });
  players.guest.send({ type: 'ack', tick: 3, digest: 'native-final', complete: 'player dead' });
  const completed = await players.host.next('completed');
  assert.deepEqual(completed, { type: 'completed', tick: 3, complete: 'player dead' });
  assert.deepEqual(await players.guest.next('completed'), completed);
  assert.equal(fixtureValue.service.stats().rooms, 0);
  await players.host.absent('error', 20);
  await players.guest.absent('error', 20);
  // A new game uses a new room and secret on the same still-open connection.
  players.host.send(CREATE);
  const newInvite = await players.host.next('joined');
  assert.notEqual(newInvite.room, players.invite.room);

// These socket regressions verify ordering and failure isolation. Synthetic
// digests are intentional here; original-SWF replay is a separate browser gate.
async function beginSolo(t, fixtureValue, options = {}) {
  const host = await connect(t, fixtureValue);
  host.send({ ...CREATE, listed: true, dropIn: true, ...options });
  const invite = await host.next('joined');
  host.send({ type: 'start' });
  const prepared = await host.next('prepare');
  assert.equal(prepared.initialPlayers, 1);
  assert.equal(prepared.replay, false);
  assert.equal(prepared.personalProgression, true);
  host.send({ type: 'ready' });
  const first = await host.next('frame');
  assert.deepEqual(first, { type: 'frame', tick: 1, p1: 0, p2: 0 });
  return { host, invite, prepared, first };
}

async function advanceSolo(host, held, count, records = new Map()) {
  let frame = held;
  for (let index = 0; index < count; index += 1) {
    records.set(frame.tick, frame);
    host.send({ type: 'ack', tick: frame.tick, digest: `native-${frame.tick}` });
    frame = await host.next('frame');
    assert.equal(frame.p2, 0);
    assert.equal(frame.join, undefined);
  }
  records.set(frame.tick, frame);
  return frame;
}

function acknowledgeSoloHost(t, host, first) {
  const records = new Map([[first.tick, first]]);
  let lastTick = first.tick, joined = false;
  const listener = (data) => {
    const frame = JSON.parse(data.toString());
    if (frame.type !== 'frame') return;
    records.set(frame.tick, frame);
    lastTick = frame.tick;
    if (frame.join) joined = true;
    if (!joined && host.ws.readyState === WebSocket.OPEN) {
      host.send({ type: 'ack', tick: frame.tick, digest: `native-${frame.tick}` });
    }
  };
  host.ws.on('message', listener);
  t.after(() => host.ws.off('message', listener));
  host.send({ type: 'ack', tick: first.tick, digest: `native-${first.tick}` });
  return { records, get lastTick() { return lastTick; } };
}

async function hostFrameAtLeast(host, tick) {
  for (;;) {
    const frame = await host.next('frame');
    if (frame.tick >= tick) return frame;
  }
}

async function syncState(client, state, timeout = 2000) {
  for (;;) {
    const message = await client.next('join_sync', timeout);
    if (message.state === state) return message;
  }
}

test('distant replay stays live beyond three seconds and only the final small gap pauses the host', async (t) => {
  const fixtureValue = await fixture32(t, { maxCatchupMs: 8000, ackTimeoutMs: 1000 });
  const { host, invite, prepared, first } = await beginSolo(t, fixtureValue, { settings: { ...SETTINGS, speed: 2 } });
  const live = acknowledgeSoloHost(t, host, first);
  await hostFrameAtLeast(host, 24);
  const guest = await connect(t, fixtureValue);
  guest.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH, character: 2 });
  await guest.next('joined');
  assert.deepEqual(await guest.next('prepare'), { ...prepared, replay: true, replayBatchMax: 8, replayBatchExtendedMax: 32 });
  const beforeLoading = live.lastTick;
  await hostFrameAtLeast(host, beforeLoading + 5);
  assert.ok(live.lastTick > beforeLoading, 'Guest native loading must leave the host running');
  const beforeReady = live.lastTick;
  const syncMessages = [];
  let barrierStarted = false;
  const observeSync = data => {
    const message = JSON.parse(data.toString());
    if (message.type === 'join_sync') {
      syncMessages.push(message);
      if (message.state === 'paused') barrierStarted = true;
    }
  };
  guest.ws.on('message', observeSync);
  t.after(() => guest.ws.off('message', observeSync));
  const replayStartedAt = performance.now();
  guest.send({ type: 'ready' });
  // Start slower than the Fast host for longer than the old three-second
  // forced pause, then catch up faster. No historical step is skipped.
  const replayDone = (async () => {
    let ticks = 0;
    for (;;) {
      const message = await guest.next(['frame', 'caught_up']);
      if (message.type === 'caught_up') return { message, ticks };
      assert.equal(message.tick, ++ticks);
      assert.deepEqual(message, live.records.get(message.tick));
      assert.equal(message.join, undefined);
      const stillDistant = performance.now() - replayStartedAt < 3200;
      if (stillDistant) assert.equal(barrierStarted, false, 'Distant replay must never freeze the host');
      await new Promise(resolve => setTimeout(resolve, stillDistant || barrierStarted ? 35 : 5));
      guest.send({ type: 'ack', tick: message.tick, digest: `native-${message.tick}` });
    }
  })();
  replayDone.catch(() => {}); // A prior assertion failure must not leave an unhandled test task.
  const [hostPause, guestPause] = await Promise.all([
    syncState(host, 'paused', 7000), syncState(guest, 'paused', 7000),
  ]);
  assert.deepEqual(hostPause, guestPause);
  assert.ok(performance.now() - replayStartedAt >= 3200);
  assert.ok(hostPause.tick >= beforeReady + 100, 'The host must keep advancing throughout distant replay');
  assert.ok(hostPause.tick - hostPause.verified <= 8, 'Only a near-target guest may pause live play');
  assert.ok(hostPause.timeoutMs > 0 && hostPause.timeoutMs <= 1500);
  const target = hostPause.tick;
  host.send({ type: 'input', seq: 0, mask: 16 });
  host.send({ type: 'input', seq: 1, mask: 0 });
  host.send({ type: 'input', seq: 2, mask: 64 });
  host.send({ type: 'input', seq: 3, mask: 0 });
  host.send({ type: 'input', seq: 4, mask: 8 });

  await hostFrameAtLeast(host, 30);

  await guest.next('joined');
  await guest.next('prepare');
  const pending = [], replayed = [];
  let timer, stopped = false, replayAcks = 0;
  const pump = () => {
    if (stopped || timer || !pending.length) return;

      timer = undefined;
      if (stopped) return;
      const frame = pending.shift();
      replayed.push(frame);
      guest.send({ type: 'ack', tick: frame.tick, digest: `native-${frame.tick}` });
      replayAcks++;
      pump();

    if (message.type === 'frame') { pending.push(message); pump(); }
    if (message.type === 'paused') { stopped = true; clearTimeout(timer); }

  assert.ok(rejected.tick > 0 && rejected.tick < live.lastTick);
  assert.ok(live.lastTick >= beforeReady + 20, 'Slow replay must not hold the live host');
  assert.equal(fixtureValue.service.catchupStats().barriers, 0);
  await host.absent('join_sync');
  const resumed = await hostFrameAtLeast(host, live.lastTick + 2);
  assert.equal(resumed.p2, 0);
  assert.equal(resumed.join, undefined);
  assert.equal(fixtureValue.service.stats().rooms, 1);
  await guest.close();

  const replacement = await connect(t, fixtureValue);
  replacement.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH, character: 2 });
  await replacement.next('joined');
  assert.deepEqual(await replacement.next('prepare'), { ...prepared, replay: true, replayBatchMax: 8, replayBatchExtendedMax: 32 }, 'Failed joining must not reset seed/settings/history');
  replacement.send({ type: 'ready' });

    const message = await replacement.next(['frame', 'caught_up']);
    if (message.type === 'caught_up') { caught = message; break; }
    assert.deepEqual(message, live.records.get(message.tick));
    replacement.send({ type: 'ack', tick: message.tick, digest: `native-${message.tick}` });

test('server-issued verified replay ACKs can exceed the control message budget without losing history checks', async (t) => {
  const messagesPerSecond = 60, messageBurst = 16;

  const { host, invite, first } = await beginSolo(t, fixtureValue, { settings: { ...SETTINGS, speed: 2 } });

      records.set(frame.tick, frame);

    return frame;

  const hostJoin = await joinedHostFrame;

  assert.equal(counters.started, 1);
  assert.equal(counters.completed, 1);
  assert.equal(counters.failed, 0);
  assert.equal(counters.verifiedReplayFrames, replayTicks);
  assert.equal(counters.barriers, 1);

test('stalled final replay handoff is bounded and automatically resumes solo play', async (t) => {
  const fixtureValue = await fixture32(t, { maxInFlightFrames: 1, maxReplayInFlightFrames: 1, ackTimeoutMs: 350 });
  const { host, invite, first } = await beginSolo(t, fixtureValue);
  const held = await advanceSolo(host, first, 2);
  let lastHostTick = held.tick;
  const autoAck = (data) => {
    const message = JSON.parse(data.toString());
    if (message.type === 'frame') {
      lastHostTick = message.tick;
      host.send({ type: 'ack', tick: message.tick, digest: `native-${message.tick}` });
    }
  };
  host.ws.on('message', autoAck);
  t.after(() => host.ws.off('message', autoAck));
  host.send({ type: 'ack', tick: held.tick, digest: `native-${held.tick}` });
  const guest = await connect(t, fixtureValue);
  guest.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH });
  await guest.next('joined');
  await guest.next('prepare');
  guest.send({ type: 'ready' });
  // Consume until the guest is one frame behind the current host. The server
  // then freezes only the final handoff; deliberately withhold the final ACK.
  for (;;) {
    const replay = await guest.next('frame');
    if (fixtureValue.service.catchupStats().barriers > 0) break;
    guest.send({ type: 'ack', tick: replay.tick, digest: `native-${replay.tick}` });
  }
  const hostAtBarrier = lastHostTick;
  const paused = await guest.next('paused');
  assert.ok(['catchup_timeout', 'catchup_barrier_timeout'].includes(paused.reason));
  assert.equal(fixtureValue.service.stats().rooms, 1);
  assert.equal(fixtureValue.service.catchupStats().failed, 1);
  let newer;
  do { newer = await host.next('frame'); } while (newer.tick <= hostAtBarrier);
  assert.ok(newer.tick > hostAtBarrier);
  assert.ok(fixtureValue.service.catchupStats().maximumBarrierMs < 1000);
});

test('batch32 converges on a persistently moving host with slower host and faster guest ACK pacing', async (t) => {
  const fixtureValue = await fixture32(t, { maxInFlightFrames: 1, maxCatchupMs: 8000, ackTimeoutMs: 2000 });
  const { host, invite, first } = await beginSolo(t, fixtureValue, { settings: { ...SETTINGS, speed: 2 } });
  const records = new Map();
  const held = await advanceSolo(host, first, 64, records);
  const timers = new Set();
  let lastHostTick = held.tick;
  // Artificial socket pacing (~22 host / ~45 guest ACKs/s), not native FPS
  // evidence. The moving target must continue throughout the replay chase.
  const acknowledgeLater = (frame) => {
    if (frame.join) return;
    const timer = setTimeout(() => {
      timers.delete(timer);
      if (host.ws.readyState === WebSocket.OPEN)
        host.send({ type: 'ack', tick: frame.tick, digest: 'native-' + frame.tick });
    }, 45);
    timers.add(timer);
  };
  const onHostFrame = data => {
    const frame = JSON.parse(data.toString());
    if (frame.type !== 'frame') return;
    records.set(frame.tick, frame); lastHostTick = frame.tick;
    acknowledgeLater(frame);
  };
  host.ws.on('message', onHostFrame);
  t.after(() => { host.ws.off('message', onHostFrame); for (const timer of timers) clearTimeout(timer); });
  const guest = await connect(t, fixtureValue);
  guest.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH });
  await guest.next('joined'); await guest.next('prepare');
  const beforeReady = lastHostTick;
  guest.send({ type: 'ready', replayBatch: 32 });
  acknowledgeLater(held);
  let verified = 0, caught;
  const sizes = [];
  for (;;) {
    const message = await guest.next(['replay_frames', 'caught_up', 'error', 'paused'], 8500);
    assert.notEqual(message.type, 'error'); assert.notEqual(message.type, 'paused');
    if (message.type === 'caught_up') { caught = message; break; }
    sizes.push(message.frames.length);
    assertReplayBatch(message, verified + 1, message.frames.length, records);
    assert.ok(message.frames.length <= 32);
    for (const frame of message.frames) {
      await new Promise(resolve => setTimeout(resolve, 22));
      guest.send({ type: 'ack', tick: frame.tick, digest: 'native-' + frame.tick });
      verified++;
    }
  }
  assert.deepEqual(sizes.slice(0, 2), [32, 32]);
  assert.ok(sizes.includes(8), 'The moving verified tip must use adaptive full groups8');
  const paused = await syncState(host, 'paused');
  assert.ok(paused.tick - paused.verified <= 8);
  assert.ok(paused.tick >= beforeReady + 20, 'Replay must not stop the moving host early');
  assert.equal(caught.tick, verified); assert.equal(caught.tick, paused.tick);
  assert.equal((await syncState(host, 'resumed')).joined, true);
  const spawn = await hostFrameAtLeast(host, caught.tick + 1);
  assert.equal(spawn.tick, caught.tick + 1); assert.equal(spawn.join.slot, 2);
  assert.deepEqual(await guest.next('frame'), spawn);
  assert.equal(fixtureValue.service.catchupStats().failed, 0);
  assert.equal(fixtureValue.service.catchupStats().completed, 1);
});

test('batch64 converges on a persistently moving host through adaptive groups8', async (t) => {
  const fixtureValue = await fixture(t, { maxInFlightFrames: 1, maxCatchupMs: 20000, ackTimeoutMs: 3000 });
  const { host, invite, first } = await beginSolo(t, fixtureValue, { settings: { ...SETTINGS, speed: 2 } });
  const records = new Map();
  const held = await advanceSolo(host, first, 128, records);
  const timers = new Set();
  let lastHostTick = held.tick;
  // Artificial socket ACK pacing verifies convergence, not native FPS or
  // the duration of one real64-frame browser task.
  const acknowledgeLater = frame => {
    if (frame.join) return;
    const timer = setTimeout(() => {
      timers.delete(timer);
      if (host.ws.readyState === WebSocket.OPEN)
        host.send({ type: 'ack', tick: frame.tick, digest: 'native-' + frame.tick });
    }, 45);
    timers.add(timer);
  };
  const onHostFrame = data => {
    const frame = JSON.parse(data.toString());
    if (frame.type !== 'frame') return;
    records.set(frame.tick, frame); lastHostTick = frame.tick;
    acknowledgeLater(frame);
  };
  host.ws.on('message', onHostFrame);
  t.after(() => { host.ws.off('message', onHostFrame); for (const timer of timers) clearTimeout(timer); });
  const guest = await connect(t, fixtureValue);
  guest.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH });
  await guest.next('joined'); await guest.next('prepare');
  const beforeReady = lastHostTick;
  guest.send({ type: 'ready', replayBatch: 64 });
  acknowledgeLater(held);
  let verified = 0, caught;
  const sizes = [];
  for (;;) {
    const message = await guest.next(['replay_frames', 'caught_up', 'error', 'paused'], 21000);
    assert.notEqual(message.type, 'error'); assert.notEqual(message.type, 'paused');
    if (message.type === 'caught_up') { caught = message; break; }
    sizes.push(message.frames.length);
    assertReplayBatch(message, verified + 1, message.frames.length, records);
    assert.ok(message.frames.length <= 64);
    for (const frame of message.frames) {
      await new Promise(resolve => setTimeout(resolve, 22));
      guest.send({ type: 'ack', tick: frame.tick, digest: 'native-' + frame.tick });
      verified++;
    }
  }
  assert.deepEqual(sizes.slice(0, 2), [64, 64]);
  assert.ok(sizes.includes(8), 'The moving verified tip must use adaptive groups8');
  const paused = await syncState(host, 'paused');
  assert.ok(paused.tick - paused.verified <= 8);
  assert.ok(paused.tick >= beforeReady + 20, 'Replay must not stop the moving host early');
  assert.equal(caught.tick, verified); assert.equal(caught.tick, paused.tick);
  assert.equal((await syncState(host, 'resumed')).joined, true);
  const spawn = await hostFrameAtLeast(host, caught.tick + 1);
  assert.equal(spawn.tick, caught.tick + 1); assert.equal(spawn.join.slot, 2);
  assert.deepEqual(await guest.next('frame'), spawn);
  assert.equal(fixtureValue.service.catchupStats().failed, 0);
  assert.equal(fixtureValue.service.catchupStats().completed, 1);
});

    assert.equal((await guest.next('prepare')).checkpointVersion, 1);
    guest.send({ type: 'ready', checkpointVersion: 1 });
    const selected = await guest.next(['frame', 'checkpoint_wait']);
    if (verifiedTicks === 49) {
      assert.deepEqual(selected, records.get(1), 'Issued tick50 alone must not select checkpoint');
      await f.host.absent('checkpoint_request', 20);
      await guest.absent('checkpoint_wait', 20);
      assert.equal(f.service.catchupStats().checkpointRequests, 0);
    } else {
      assert.equal(selected.type, 'checkpoint_wait', 'Verified tick50 must select checkpoint');
      const request = await f.host.next('checkpoint_request');
      assert.equal(request.id, selected.id);
      assert.equal(request.minimumTick, 51);
      assert.equal(f.service.catchupStats().checkpointRequests, 1);
      await guest.absent('frame', 20);
    }
    const next = await advanceSolo(f.host, held, 1, records);
    assert.equal(next.tick, verifiedTicks + 2, 'Both selected paths preserve ongoing solo host steps');
    assert.equal(f.service.catchupStats().barriers, 0);
    assert.equal(f.service.catchupStats().verifiedReplayFrames, 0);
    if (verifiedTicks === 49) {
      await f.host.absent('checkpoint_request', 20);
      assert.equal(f.service.catchupStats().checkpointRequests, 0, 'A selected short replay does not switch after crossing the threshold');
    }
  });
});

async function checkpointFixture(t, options = {}) {
  const fx = await checkpointCapableSolo(t, { minCheckpointHistoryTicks: 1, ...options });
  const { host, invite } = fx;
  let held = await advanceSolo(host, fx.first, 25);
  const guest = await connect(t, fx);
  guest.send({ type: 'join_open', room: invite.room, swfHash: SWF_HASH });
  await guest.next('joined');
  assert.equal((await guest.next('prepare')).checkpointVersion, 1);
  guest.send({ type: 'ready', checkpointVersion: 1 });
  const waiting = await guest.next('checkpoint_wait');
  const request = await host.next('checkpoint_request');
  assert.equal(waiting.id, request.id);
  held = await advanceSolo(host, held, 25);
  assert.equal(held.tick, 51);
  const bytes = Buffer.alloc(40000, 37);
  const header = { type: 'checkpoint_begin', id: request.id, version: 1, tick: 50,
    digest: 'native-50', encoding: 'plain', bytes: bytes.length, uncompressedBytes: bytes.length,
    sha256: createHash('sha256').update(bytes).digest('hex'), chunks: 2 };
  return { ...fx, host, guest, held, bytes, header };
}

async function sendCheckpoint(f, bytes = f.bytes) {
  f.host.send(f.header);
  for (let offset = 0, index = 0; offset < bytes.length; offset += 32768, index++) {
    assert.equal((await f.host.next('checkpoint_credit')).index, index);
    f.host.send({ type: 'checkpoint_chunk', id: f.header.id, index, data: bytes.subarray(offset, offset + 32768).toString('base64') });
  }
  assert.equal((await f.host.next('checkpoint_credit')).index, f.header.chunks);
  f.host.send({ type: 'checkpoint_end', id: f.header.id });
}

test('checkpoint attestation skips no synthetic ACKs and verifies the real ordered tail before spawn', async t => {
  const f = await checkpointFixture(t);
  assert.equal(f.service.catchupStats().barriers, 0, 'Capturing/uploading must leave the solo host live');
  await sendCheckpoint(f);
  assert.deepEqual(await f.guest.next('checkpoint_begin'), f.header);
  assert.equal((await f.guest.next('checkpoint_chunk')).index, 0);
  assert.equal((await f.guest.next('checkpoint_chunk')).index, 1);
  await f.guest.next('checkpoint_end');
  assert.equal(f.service.catchupStats().verifiedReplayFrames, 0);
  f.guest.send({ type: 'checkpoint_applied', id: f.header.id, tick: 50, digest: 'native-50', sha256: f.header.sha256 });
  await f.host.next('checkpoint_cancel');
  assert.equal((await syncState(f.host, 'paused')).tick, 51);
  f.host.send({ type: 'ack', tick: 51, digest: 'native-51' });
  assert.deepEqual(await f.guest.next('frame'), f.held);
  f.guest.send({ type: 'ack', tick: 51, digest: 'native-51' });
  assert.equal((await f.guest.next('caught_up')).tick, 51);
  const join = await f.guest.next('frame');
  assert.equal(join.tick, 52); assert.equal(join.join.slot, 2);
  assert.deepEqual(await f.host.next('frame'), join);
  assert.equal(f.service.catchupStats().restoredCheckpoints, 1);
  assert.equal(f.service.catchupStats().verifiedReplayFrames, 1);
});

test('corrupt checkpoint payload cancels only the guest and never freezes the solo host', async t => {
  const f = await checkpointFixture(t);
  const bad = Buffer.from(f.bytes); bad[0] ^= 1;
  await sendCheckpoint(f, bad);
  assert.equal((await f.guest.next('paused')).reason, 'invalid_checkpoint_end');
  assert.equal((await f.host.next('checkpoint_cancel')).id, f.header.id);
  assert.equal(f.service.catchupStats().restoredCheckpoints, 0);
  assert.equal(f.service.catchupStats().barriers, 0);
  assert.equal(f.service.stats().rooms, 1);
  assert.equal((await advanceSolo(f.host, f.held, 1)).tick, 52);
  await f.guest.absent('checkpoint_begin', 20);
});

test('unavailable optional checkpoint explicitly falls back to verified history from tick one', async t => {
  const f = await checkpointFixture(t);
  f.host.send({ type: 'checkpoint_unavailable', id: f.header.id });
  assert.equal((await f.guest.next('checkpoint_fallback')).id, f.header.id);
  await f.host.next('checkpoint_cancel');
  assert.equal((await f.guest.next('frame')).tick, 1);
  assert.equal(f.service.catchupStats().restoredCheckpoints, 0);
  assert.equal(f.service.catchupStats().checkpointFallbacks, 1);
  assert.equal(f.service.catchupStats().barriers, 0);
  assert.equal((await advanceSolo(f.host, f.held, 1)).tick, 52);
});

test('maximum checkpoint transfer keeps the host live while a real guest socket is paused', async t => {
  const f = await checkpointFixture(t, { ackTimeoutMs: 5000 });
  f.bytes = Buffer.alloc(4 * 1024 * 1024);
  for (let i = 0; i < f.bytes.length; i++) f.bytes[i] = i % 251;
  f.header = { ...f.header, bytes: f.bytes.length, uncompressedBytes: f.bytes.length,
    chunks: f.bytes.length / 32768, sha256: createHash('sha256').update(f.bytes).digest('hex') };
  f.guest.ws.pause();
  const resume = setTimeout(() => f.guest.ws.resume(), 300);
  t.after(() => { clearTimeout(resume); f.guest.ws.resume(); });
  const upload = sendCheckpoint(f);
  const live = await advanceSolo(f.host, f.held, 5);
  assert.equal(live.tick, 56);
  assert.equal(live.join, undefined);
  assert.equal(f.service.catchupStats().barriers, 0);
  await upload;
  assert.deepEqual(await f.guest.next('checkpoint_begin'), f.header);
  const parts = [];
  for (let index = 0; index < f.header.chunks; index++) {
    const part = await f.guest.next('checkpoint_chunk');
    assert.equal(part.index, index); assert.equal(part.id, f.header.id);
    parts.push(Buffer.from(part.data, 'base64'));
  }
  await f.guest.next('checkpoint_end');
  assert.deepEqual(Buffer.concat(parts), f.bytes);
  assert.equal(f.service.catchupStats().verifiedReplayFrames, 0);
  assert.equal(f.service.catchupStats().restoredCheckpoints, 0);
  assert.equal(f.service.catchupStats().failed, 0);
  assert.equal((await advanceSolo(f.host, live, 1)).tick, 57);
  await f.host.absent('join_sync', 20);
});
